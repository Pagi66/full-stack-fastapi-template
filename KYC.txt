KYC Verification & Account Settings Implementation

1. Database Schema for KYC & User Profiles

```sql
-- Extended user profile for KYC data collection
ALTER TABLE users ADD COLUMN IF NOT EXISTS 
    kyc_status VARCHAR(20) DEFAULT 'pending' CHECK (kyc_status IN ('pending', 'under_review', 'approved', 'rejected'));

ALTER TABLE users ADD COLUMN IF NOT EXISTS 
    kyc_submitted_at TIMESTAMPTZ,
    kyc_approved_at TIMESTAMPTZ,
    kyc_rejected_reason TEXT;

-- KYC document storage
CREATE TABLE kyc_documents (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    document_type VARCHAR(50) CHECK (document_type IN ('passport', 'drivers_license', 'national_id', 'proof_of_address')),
    front_image_url TEXT,
    back_image_url TEXT,
    verified BOOLEAN DEFAULT FALSE,
    verified_by UUID REFERENCES users(id),
    verified_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- User profile extensions for data collection
CREATE TABLE user_profiles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id) UNIQUE,
    legal_first_name VARCHAR(100),
    legal_last_name VARCHAR(100),
    date_of_birth DATE,
    phone_number VARCHAR(20),
    address_line_1 TEXT,
    address_line_2 TEXT,
    city VARCHAR(100),
    state VARCHAR(100),
    postal_code VARCHAR(20),
    country VARCHAR(100),
    tax_id_number VARCHAR(50), -- SSN/Tax ID
    occupation VARCHAR(100),
    source_of_funds VARCHAR(100),
    risk_assessment_score INTEGER DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);
```

5. Backend KYC Verification API

```python
# File: backend/app/api/routes/kyc.py
@router.post("/submit-kyc")
async def submit_kyc_information(
    kyc_data: KYCSchema,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Submit KYC information for verification"""
    
    # Validate age requirement (18+)
    age = calculate_age(kyc_data.date_of_birth)
    if age < 18:
        raise HTTPException(status_code=400, detail="Must be 18 years or older")
    
    # Store KYC data
    user_profile = db.exec(
        select(UserProfile).where(UserProfile.user_id == current_user.id)
    ).first()
    
    if not user_profile:
        user_profile = UserProfile(user_id=current_user.id)
    
    # Update profile with KYC data
    for field, value in kyc_data.dict().items():
        setattr(user_profile, field, value)
    
    # Update user KYC status
    current_user.kyc_status = "under_review"
    current_user.kyc_submitted_at = datetime.utcnow()
    
    db.add(user_profile)
    db.add(current_user)
    db.commit()
    
    # Trigger admin notification
    await notify_admins_kyc_submission(current_user.id)
    
    return {"message": "KYC information submitted for review"}

@router.post("/upload-document")
async def upload_kyc_document(
    document_type: str,
    file: UploadFile,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Upload KYC document for verification"""
    
    # Validate file type and size
    if not allowed_file_type(file.filename):
        raise HTTPException(status_code=400, detail="Invalid file type")
    
    if file.size > 10 * 1024 * 1024:  # 10MB limit
        raise HTTPException(status_code=400, detail="File too large")
    
    # Upload to secure storage
    file_url = await upload_to_secure_storage(file)
    
    # Store document record
    document = KYCDocument(
        user_id=current_user.id,
        document_type=document_type,
        front_image_url=file_url,
        created_at=datetime.utcnow()
    )
    
    db.add(document)
    db.commit()
    
    return {"message": "Document uploaded successfully", "document_id": document.id}
```

```typescript
export const KYCReview = () => {
```
